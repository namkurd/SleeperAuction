# Runs the DTF Club auction dashboard: fetches new Sleeper drafts, rebuilds the data, publishes the site.
# Save this file as .github/workflows/update.yml (any file name inside .github/workflows works).

name: Update auction data and publish site

on:
  schedule:
    - cron: "17 */6 * * *"        # every 6 hours
  workflow_dispatch:
    inputs:
      refresh:
        description: "Re-download every Sleeper season (normally not needed)"
        type: boolean
        default: false
  push:
    branches: [main]
    paths:
      - "site/**"
      - "config/**"
      - "scripts/**"
      - "data/history_2012_2020.csv"
      - ".github/workflows/**"

permissions:
  contents: write     # commit refreshed data
  actions: write      # keep the schedule from being auto-disabled
  pages: write
  id-token: write

concurrency:
  group: dtf-auction
  cancel-in-progress: false

jobs:
  update:
    runs-on: ubuntu-latest
    outputs:
      changed: ${{ steps.commit.outputs.changed }}
    steps:
      - uses: actions/checkout@v4
        with:
          ref: main
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Fetch new Sleeper drafts and rebuild data
        run: python scripts/update_data.py ${{ inputs.refresh && '--refresh' || '' }}

      - name: Sanity-check the data
        run: python scripts/check_data.py

      - name: Commit if anything changed
        id: commit
        run: |
          git config user.name  "github-actions[bot]"
          git config user.email "41898282+github-actions[bot]@users.noreply.github.com"
          if [ -n "$(git status --porcelain data site)" ]; then
            git add data site
            git commit -m "Update auction data"
            git pull --rebase origin main
            git push origin HEAD:main
            echo "changed=true" >> "$GITHUB_OUTPUT"
          else
            echo "changed=false" >> "$GITHUB_OUTPUT"
          fi

      # GitHub switches scheduled workflows off after 60 days without repository activity, and the
      # NFL off-season is longer than that. Two belts and braces:
      - name: Keep the schedule alive
        if: github.event_name == 'schedule'
        env:
          GH_TOKEN: ${{ github.token }}
        run: |
          gh workflow enable "$GITHUB_WORKFLOW" || true
          last=$(git log -1 --format=%ct); now=$(date +%s)
          if [ $(( now - last )) -gt $(( 45 * 86400 )) ]; then
            date -u +%F > data/heartbeat.txt
            git add data/heartbeat.txt
            git commit -m "Heartbeat: keep scheduled updates enabled"
            git pull --rebase origin main
            git push origin HEAD:main
          fi

  deploy:
    needs: update
    # publish when data changed, or when a person pushed / clicked "Run workflow"
    if: needs.update.outputs.changed == 'true' || github.event_name != 'schedule'
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4
        with:
          ref: main
      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: site
      - id: deployment
        uses: actions/deploy-pages@v4
